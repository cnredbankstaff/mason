  <script>
    // ===== CONFIG & SDK INIT =====
    const defaultConfig = {
      game_title: 'Beat Runner',
      background_color: '#030712',
      surface_color: '#1e1b4b',
      text_color: '#c4b5fd',
      primary_action_color: '#7c3aed',
      secondary_action_color: '#6366f1'
    };

    window.elementSdk.init({
      defaultConfig,
      onConfigChange: async (config) => {
        const title = config.game_title || defaultConfig.game_title;
        document.getElementById('titleText').textContent = title;
        document.title = title;
        document.body.style.backgroundColor = config.background_color || defaultConfig.background_color;
      },
      mapToCapabilities: (config) => ({
        recolorables: [
          { get: () => config.background_color || defaultConfig.background_color, set: (v) => { config.background_color = v; window.elementSdk.setConfig({ background_color: v }); } },
          { get: () => config.surface_color || defaultConfig.surface_color, set: (v) => { config.surface_color = v; window.elementSdk.setConfig({ surface_color: v }); } },
          { get: () => config.text_color || defaultConfig.text_color, set: (v) => { config.text_color = v; window.elementSdk.setConfig({ text_color: v }); } },
          { get: () => config.primary_action_color || defaultConfig.primary_action_color, set: (v) => { config.primary_action_color = v; window.elementSdk.setConfig({ primary_action_color: v }); } },
          { get: () => config.secondary_action_color || defaultConfig.secondary_action_color, set: (v) => { config.secondary_action_color = v; window.elementSdk.setConfig({ secondary_action_color: v }); } }
        ],
        borderables: [],
        fontEditable: undefined,
        fontSizeable: undefined
      }),
      mapToEditPanelValues: (config) => new Map([
        ['game_title', config.game_title || defaultConfig.game_title]
      ])
    });

    // ===== AUDIO ENGINE =====
    let audioCtx, analyser, source, audioBuffer, audioSource;
    let freqData, timeData;
    const FFT_SIZE = 256;

    async function loadAudio(file) {
      audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      analyser = audioCtx.createAnalyser();
      analyser.fftSize = FFT_SIZE;
      analyser.smoothingTimeConstant = 0.8;
      freqData = new Uint8Array(analyser.frequencyBinCount);
      timeData = new Uint8Array(analyser.fftSize);

      const arrayBuffer = await file.arrayBuffer();
      audioBuffer = await audioCtx.decodeAudioData(arrayBuffer);
      document.getElementById('songName').textContent = file.name.replace(/\.[^.]+$/, '');
      document.getElementById('songInfo').classList.remove('hidden');
      document.getElementById('themeSelect').classList.remove('hidden');
    }

    function playAudio() {
      if (audioSource) { try { audioSource.stop(); } catch(e) {} }
      audioSource = audioCtx.createBufferSource();
      audioSource.buffer = audioBuffer;
      audioSource.connect(analyser);
      analyser.connect(audioCtx.destination);
      audioSource.start();
      audioSource.onended = () => { if (!game.paused) endGame(); };
    }

    function stopAudio() {
      if (audioSource) { try { audioSource.stop(); } catch(e) {} }
    }

    function getAudioData() {
      if (!analyser) return { bass: 0, mid: 0, treble: 0, volume: 0, beat: false };
      analyser.getByteFrequencyData(freqData);
      analyser.getByteTimeDomainData(timeData);
      const bins = freqData.length;
      let bass = 0, mid = 0, treble = 0;
      for (let i = 0; i < bins * 0.15; i++) bass += freqData[i];
      for (let i = Math.floor(bins * 0.15); i < bins * 0.5; i++) mid += freqData[i];
      for (let i = Math.floor(bins * 0.5); i < bins; i++) treble += freqData[i];
      bass = bass / (bins * 0.15) / 255;
      mid = mid / (bins * 0.35) / 255;
      treble = treble / (bins * 0.5) / 255;
      const volume = (bass + mid + treble) / 3;
      const beat = bass > 0.7;
      return { bass, mid, treble, volume, beat };
    }

    // ===== GAME STATE =====
    const game = {
      running: false, paused: false, score: 0, combo: 1, maxCombo: 1,
      player: { x: 100, y: 300, vy: 0, w: 30, h: 30, grounded: false, alive: true },
      obstacles: [], particles: [], platforms: [], bgParticles: [],
      keys: {}, theme: 'neon', speed: 4, gravity: 0.6, jumpForce: -12,
      spawnTimer: 0, difficultyTimer: 0, comboTimer: 0
    };

    const themes = {
      neon: { bg: '#0a0015', colors: ['#ff00ff', '#00ffff', '#ff6600', '#ffff00', '#8b5cf6'], ground: '#1a0030' },
      space: { bg: '#000011', colors: ['#4488ff', '#88ccff', '#ffffff', '#6644ff', '#00aaff'], ground: '#000033' },
      cyber: { bg: '#001111', colors: ['#00ffcc', '#00ff88', '#88ffff', '#ffcc00', '#ff4488'], ground: '#002222' },
      forest: { bg: '#001a00', colors: ['#00ff44', '#44ff88', '#88ffcc', '#ffdd00', '#ff8800'], ground: '#002200' }
    };

    // ===== GAME CANVAS =====
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d');

    function resizeCanvas() {
      canvas.width = canvas.clientWidth * window.devicePixelRatio;
      canvas.height = canvas.clientHeight * window.devicePixelRatio;
      ctx.scale(window.devicePixelRatio, window.devicePixelRatio);
    }

    // ===== MENU BACKGROUND =====
    const menuCanvas = document.getElementById('menuBg');
    const menuCtx = menuCanvas.getContext('2d');
    let menuParticles = [];

    function initMenuBg() {
      menuCanvas.width = menuCanvas.clientWidth * window.devicePixelRatio;
      menuCanvas.height = menuCanvas.clientHeight * window.devicePixelRatio;
      menuCtx.scale(window.devicePixelRatio, window.devicePixelRatio);
      menuParticles = Array.from({ length: 60 }, () => ({
        x: Math.random() * menuCanvas.clientWidth,
        y: Math.random() * menuCanvas.clientHeight,
        r: Math.random() * 2 + 1,
        vx: (Math.random() - 0.5) * 0.5,
        vy: (Math.random() - 0.5) * 0.5,
        color: ['#8b5cf6', '#6366f1', '#ec4899', '#06b6d4'][Math.floor(Math.random() * 4)]
      }));
      animateMenu();
    }

    function animateMenu() {
      if (game.running) return;
      const w = menuCanvas.clientWidth, h = menuCanvas.clientHeight;
      menuCtx.clearRect(0, 0, w, h);
      menuParticles.forEach(p => {
        p.x += p.vx; p.y += p.vy;
        if (p.x < 0 || p.x > w) p.vx *= -1;
        if (p.y < 0 || p.y > h) p.vy *= -1;
        menuCtx.beginPath();
        menuCtx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
        menuCtx.fillStyle = p.color;
        menuCtx.globalAlpha = 0.6;
        menuCtx.fill();
        menuCtx.globalAlpha = 1;
      });
      requestAnimationFrame(animateMenu);
    }

    // ===== EQ BARS =====
    const eqContainer = document.getElementById('eqBars');
    const barCount = 32;
    for (let i = 0; i < barCount; i++) {
      const bar = document.createElement('div');
      bar.className = 'eq-bar';
      bar.style.cssText = `width: ${100/barCount}%; height: 4px; background: linear-gradient(to top, #8b5cf6, #ec4899); border-radius: 2px 2px 0 0; transition: height 0.05s;`;
      eqContainer.appendChild(bar);
    }

    function updateEQ() {
      if (!analyser) return;
      analyser.getByteFrequencyData(freqData);
      const bars = eqContainer.children;
      const step = Math.floor(freqData.length / barCount);
      for (let i = 0; i < barCount; i++) {
        const val = freqData[i * step] / 255;
        bars[i].style.height = (val * 80 + 4) + 'px';
      }
    }

    // ===== GAME LOGIC =====
    function startGame() {
      resizeCanvas();
      game.running = true; game.paused = false; game.score = 0; game.combo = 1; game.maxCombo = 1;
      game.player = { x: 100, y: canvas.clientHeight - 100, vy: 0, w: 28, h: 28, grounded: false, alive: true };
      game.obstacles = []; game.particles = []; game.platforms = []; game.bgParticles = [];
      game.speed = 4; game.spawnTimer = 0; game.difficultyTimer = 0; game.comboTimer = 0;

      // Init bg particles
      for (let i = 0; i < 40; i++) {
        game.bgParticles.push({
          x: Math.random() * canvas.clientWidth,
          y: Math.random() * canvas.clientHeight,
          r: Math.random() * 3 + 1,
          speed: Math.random() * 1 + 0.5
        });
      }

      document.getElementById('menu').classList.add('hidden');
      document.getElementById('gameCanvas').classList.remove('hidden');
      document.getElementById('hud').classList.remove('hidden');
      document.getElementById('eqBars').classList.remove('hidden');
      document.getElementById('gameOver').classList.add('hidden');
      document.getElementById('pauseOverlay').classList.add('hidden');

      playAudio();
      requestAnimationFrame(gameLoop);
    }

    function endGame() {
      game.running = false; game.player.alive = false;
      stopAudio();
      document.getElementById('finalScore').textContent = game.score;
      document.getElementById('finalCombo').textContent = 'x' + game.maxCombo;
      document.getElementById('gameOver').classList.remove('hidden');
    }

    let lastTime = 0;
    function gameLoop(time) {
      if (!game.running) return;
      if (game.paused) { requestAnimationFrame(gameLoop); return; }
      const dt = Math.min((time - lastTime) / 16.67, 2);
      lastTime = time;

      const audio = getAudioData();
      const w = canvas.clientWidth, h = canvas.clientHeight;
      const theme = themes[game.theme];
      const groundY = h - 60;

      // Update speed based on audio
      game.speed = 4 + audio.volume * 6;
      game.spawnTimer += dt;
      game.difficultyTimer += dt;
      game.comboTimer += dt;

      // Score
      if (game.comboTimer > 30) { game.comboTimer = 0; game.score += game.combo; }

      // Spawn obstacles synced to beat
      if (audio.beat && game.spawnTimer > 20) {
        game.spawnTimer = 0;
        const type = Math.random();
        if (type < 0.5) {
          game.obstacles.push({ x: w + 50, y: groundY - 25 - Math.random() * 20, w: 25 + audio.bass * 20, h: 25 + audio.bass * 30, color: theme.colors[Math.floor(Math.random() * theme.colors.length)] });
        } else {
          game.platforms.push({ x: w + 50, y: groundY - 80 - Math.random() * 100, w: 80 + audio.mid * 60, h: 12, color: theme.colors[Math.floor(Math.random() * theme.colors.length)] });
        }
      }

      // Spawn on treble
      if (audio.treble > 0.5 && game.spawnTimer > 10) {
        game.spawnTimer = 5;
        game.obstacles.push({ x: w + 50, y: groundY - 60 - Math.random() * 80, w: 18, h: 18, color: theme.colors[2], floating: true, oy: groundY - 60 - Math.random() * 80, t: 0 });
      }

      // Player input
      const p = game.player;
      if ((game.keys['ArrowUp'] || game.keys['KeyW'] || game.keys['Space']) && p.grounded) {
        p.vy = game.jumpForce - audio.bass * 4;
        p.grounded = false;
        spawnParticles(p.x, p.y + p.h, 5, theme.colors[0]);
      }
      if (game.keys['ArrowDown'] || game.keys['KeyS']) { p.vy += 0.5; }

      // Physics
      p.vy += game.gravity * dt;
      p.y += p.vy * dt;
      p.grounded = false;

      // Ground collision
      if (p.y + p.h >= groundY) { p.y = groundY - p.h; p.vy = 0; p.grounded = true; }

      // Platform collision
      game.platforms.forEach(pl => {
        if (p.vy > 0 && p.x + p.w > pl.x && p.x < pl.x + pl.w && p.y + p.h >= pl.y && p.y + p.h <= pl.y + pl.h + 10) {
          p.y = pl.y - p.h; p.vy = 0; p.grounded = true;
        }
      });

      // Move obstacles & platforms
      game.obstacles = game.obstacles.filter(o => {
        o.x -= game.speed * dt;
        if (o.floating) { o.t += 0.05; o.y = o.oy + Math.sin(o.t) * 20; }
        return o.x > -60;
      });
      game.platforms = game.platforms.filter(pl => { pl.x -= game.speed * dt; return pl.x > -200; });

      // Collision detection
      game.obstacles.forEach(o => {
        if (p.x + p.w > o.x + 4 && p.x < o.x + o.w - 4 && p.y + p.h > o.y + 4 && p.y < o.y + o.h - 4) {
          endGame();
        }
      });

      // Combo from near misses
      game.obstacles.forEach(o => {
        if (!o.passed && o.x + o.w < p.x) {
          o.passed = true;
          game.combo++;
          game.maxCombo = Math.max(game.maxCombo, game.combo);
          game.score += 10 * game.combo;
          spawnParticles(p.x, p.y, 3, '#ffd700');
        }
      });

      // Background particles
      game.bgParticles.forEach(bp => {
        bp.x -= bp.speed * game.speed * dt * 0.3;
        if (bp.x < 0) { bp.x = w; bp.y = Math.random() * h; }
      });

      // Effect particles
      game.particles = game.particles.filter(pa => { pa.life -= dt; pa.x += pa.vx * dt; pa.y += pa.vy * dt; pa.vy += 0.2; return pa.life > 0; });

      // ===== RENDER =====
      ctx.clearRect(0, 0, w, h);

      // Background
      const grad = ctx.createLinearGradient(0, 0, 0, h);
      grad.addColorStop(0, theme.bg);
      grad.addColorStop(1, theme.ground);
      ctx.fillStyle = grad;
      ctx.fillRect(0, 0, w, h);

      // Bg particles (react to audio)
      game.bgParticles.forEach(bp => {
        ctx.beginPath();
        ctx.arc(bp.x, bp.y, bp.r + audio.volume * 3, 0, Math.PI * 2);
        ctx.fillStyle = theme.colors[0] + '44';
        ctx.fill();
      });

      // Ground line
      ctx.fillStyle = theme.colors[0] + '33';
      ctx.fillRect(0, groundY, w, 2);
      // Ground glow on beat
      if (audio.beat) {
        ctx.shadowColor = theme.colors[0];
        ctx.shadowBlur = 20;
        ctx.fillRect(0, groundY, w, 2);
        ctx.shadowBlur = 0;
      }

      // Waveform
      ctx.beginPath();
      ctx.strokeStyle = theme.colors[1] + '55';
      ctx.lineWidth = 1.5;
      const sliceW = w / timeData.length;
      for (let i = 0; i < timeData.length; i++) {
        const v = timeData[i] / 128.0;
        const y2 = (v * h / 4) + h * 0.1;
        if (i === 0) ctx.moveTo(0, y2); else ctx.lineTo(i * sliceW, y2);
      }
      ctx.stroke();

      // Platforms
      game.platforms.forEach(pl => {
        ctx.fillStyle = pl.color + 'cc';
        ctx.shadowColor = pl.color;
        ctx.shadowBlur = audio.volume * 15;
        ctx.fillRect(pl.x, pl.y, pl.w, pl.h);
        ctx.shadowBlur = 0;
      });

      // Obstacles
      game.obstacles.forEach(o => {
        ctx.save();
        ctx.translate(o.x + o.w/2, o.y + o.h/2);
        ctx.rotate(Date.now() * 0.003);
        ctx.fillStyle = o.color;
        ctx.shadowColor = o.color;
        ctx.shadowBlur = 10 + audio.bass * 20;
        ctx.fillRect(-o.w/2, -o.h/2, o.w, o.h);
        ctx.shadowBlur = 0;
        ctx.restore();
      });

      // Player
      ctx.save();
      const scale = 1 + audio.bass * 0.15;
      ctx.translate(p.x + p.w/2, p.y + p.h/2);
      ctx.scale(scale, scale);
      ctx.fillStyle = '#fff';
      ctx.shadowColor = theme.colors[0];
      ctx.shadowBlur = 15 + audio.volume * 25;
      ctx.fillRect(-p.w/2, -p.h/2, p.w, p.h);
      ctx.shadowBlur = 0;
      // Inner glow
      ctx.fillStyle = theme.colors[0];
      ctx.fillRect(-p.w/2 + 4, -p.h/2 + 4, p.w - 8, p.h - 8);
      ctx.restore();

      // Particles
      game.particles.forEach(pa => {
        ctx.globalAlpha = pa.life / pa.maxLife;
        ctx.fillStyle = pa.color;
        ctx.beginPath();
        ctx.arc(pa.x, pa.y, pa.r, 0, Math.PI * 2);
        ctx.fill();
      });
      ctx.globalAlpha = 1;

      // Beat flash
      if (audio.beat) {
        ctx.fillStyle = theme.colors[0] + '08';
        ctx.fillRect(0, 0, w, h);
      }

      // Update HUD
